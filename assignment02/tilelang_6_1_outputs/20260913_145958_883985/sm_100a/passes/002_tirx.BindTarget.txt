@I.ir_module
class Module:
    @T.prim_func
    def gemm(A_handle: T.handle, B_handle: T.handle, C_handle: T.handle):
        T.func_attr({"target": T.target({"arch": "sm_100a", "host": {"keys": ["cpu"], "kind": "c", "tag": ""}, "keys": ["cuda", "gpu"], "kind": "cuda", "max_num_threads": 1024, "tag": "", "thread_warp_size": 32})})
        A = T.match_buffer(A_handle, (1024, 1024), "bfloat16", strides=(1024, 1))
        B = T.match_buffer(B_handle, (1024, 1024), "bfloat16", strides=(1024, 1))
        C = T.match_buffer(C_handle, (1024, 1024), strides=(1024, 1))
        # with T.sblock("root"):
        for bx in T.thread_binding(8, thread="blockIdx.x"):
            for by in T.thread_binding(8, thread="blockIdx.y"):
                for tx in T.thread_binding(128, thread="threadIdx.x"):
                    for ty in T.thread_binding(1, thread="threadIdx.y"):
                        for tz in T.thread_binding(1, thread="threadIdx.z"):
                            with T.sblock("tilelang_root"):
                                T.reads()
                                T.writes()
                                A_shared = T.sblock_alloc_buffer((128, 64), "bfloat16", scope="shared.dyn")
                                B_shared = T.sblock_alloc_buffer((64, 128), "bfloat16", scope="shared.dyn")
                                C_local = T.sblock_alloc_buffer((128, 128), scope="local.fragment")
                                T.fill(T.region(C_local[0, 0], 2, 128, 128), 0)
                                for k in T.serial(16, annotations={"num_stages": 3}):
                                    T.copy(T.region(A[by * 128, k * 64], 1, 128, 64), T.region(A_shared[0, 0], 2, 128, 64))
                                    T.copy(T.region(B[k * 64, bx * 128], 1, 64, 128), T.region(B_shared[0, 0], 2, 64, 128))
                                    T.gemm(T.region(A_shared[0, 0], 1, 128, 64), T.region(B_shared[0, 0], 1, 64, 128), T.region(C_local[0, 0], 3, 128, 128), T.bool(False), T.bool(False), 128, 128, 64, 0, T.bool(False), 64, 128, 0, 0, 1, 0, 0, 0, 0)
                                T.copy(T.region(C_local[0, 0], 1, 128, 128), T.region(C[by * 128, bx * 128], 2, 128, 128))
