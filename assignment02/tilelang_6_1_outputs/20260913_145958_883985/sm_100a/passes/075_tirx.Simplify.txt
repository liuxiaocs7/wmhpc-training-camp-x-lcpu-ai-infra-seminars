        A_shared: T.handle("bfloat16", "shared.dyn") = T.handle_add_byte_offset(buf_dyn_shmem.data, 0)
        B_shared: T.handle("bfloat16", "shared.dyn") = T.handle_add_byte_offset(buf_dyn_shmem.data, 49152)
        mbarrier = T.alloc_buffer((6,), "uint64", scope="shared.barrier")
        C_local = T.alloc_buffer((128,), scope="local")
        by = T.launch_thread("blockIdx.y", 8)
        tx = T.launch_thread("threadIdx.x", 256)
        if T.tl_shuffle_elect(0):
            T.prefetch_tma_descriptor(A_desc)
            T.prefetch_tma_descriptor(B_desc)
        ty = T.launch_thread("threadIdx.y", 1)
        tz = T.launch_thread("threadIdx.z", 1)
        mbarrier_1 = T.decl_buffer((6,), "uint64", data=mbarrier.data, scope="shared.barrier")

            T.ptx_init_barrier_thread_count(mbarrier_1[1], 1)
            T.ptx_init_barrier_thread_count(mbarrier_1[2], 1)
            T.ptx_init_barrier_thread_count(mbarrier_1[3], 128)
            T.ptx_init_barrier_thread_count(mbarrier_1[4], 128)
            T.ptx_init_barrier_thread_count(mbarrier_1[5], 128)
        T.ptx_fence_barrier_init()
        T.tvm_storage_sync("shared")
        T.attr([128, 128], "kWarpSpecializationScope", 0)
        if tx < 128:
            T.set_max_nreg(24, 0)
            for k in range(16):
                T.mbarrier_wait_parity(mbarrier_1[T.truncmod(k, 3) + 3], T.bitwise_xor(T.Div(T.truncmod(k, 6), 3), 1))
                if T.tl_shuffle_elect(128):

                    A_shared_1 = T.Buffer((k % 3 * 8192 + 1,), "bfloat16", data=A_shared, scope="shared.dyn")

                    B_shared_1 = T.Buffer((k % 3 * 8192 + 1,), "bfloat16", data=B_shared, scope="shared.dyn")

            for i in T.unroll(32):

                with T.attr(0, "lexical_alloc_scope", 1):

                    B_local = T.alloc_buffer((32,), "bfloat16", scope="local")
                    for ki in range(4):
                        for i in T.unroll(4):

                        for i in T.unroll(4):
                            B_shared_1 = T.Buffer((k % 3 * 8192 + tx % 128 // 64 * 4096 + ki * 1024 + tx % 16 // 8 * 512 + (tx % 16 * 64 + (tx % 8 // 4 + i // 2) % 2 * 32 + (tx % 4 // 2 + i % 2) % 2 * 16 + (tx % 32 // 16 + tx % 2) % 2 * 8) % 512 + 1,), "bfloat16", data=B_shared, scope="shared.dyn")

                            T.ptx_ldmatrix(T.bool(True), 4, T.address_of(B_shared_1[T.truncmod(k, 3) * 8192 + T.shift_right(T.bitwise_and(tx, 127), 6) * 4096 + ki * 1024 + T.shift_right(T.bitwise_and(tx, 15), 3) * 512 + T.bitwise_and(T.bitwise_and(tx, 15) * 64 + T.bitwise_and(T.shift_right(T.bitwise_and(tx, 7), 2) + T.shift_right(i, 1), 1) * 32 + T.bitwise_and(T.shift_right(T.bitwise_and(tx, 3), 1) + T.bitwise_and(i, 1), 1) * 16 + T.bitwise_and(T.shift_right(T.bitwise_and(tx, 31), 4) + T.bitwise_and(tx, 1), 1) * 8, 511)]), T.address_of(B_local_1[i * 8]))
                        for i, j in T.grid(4, 4):
                            T.ptx_mma("float32", "m16n8k16", "row", "col", "bf16", "bf16", "fp32", A_local.data, i * 8, B_local.data, j * 8, C_local.data, i * 32 + j * 8, T.bool(False))
                            T.ptx_mma("float32", "m16n8k16", "row", "col", "bf16", "bf16", "fp32", A_local.data, i * 8, B_local.data, j * 8 + 4, C_local.data, i * 32 + j * 8 + 4, T.bool(False))
                T.ptx_arrive_barrier(mbarrier_1[T.truncmod(k, 3) + 3])
            for i in T.unroll(64):
                C_1[by * 131072 + T.shift_right(T.bitwise_and(tx, 63), 5) * 65536 + T.shift_right(i, 4) * 16384 + T.bitwise_and(i, 1) * 8192 + T.shift_right(T.bitwise_and(tx, 31), 2) * 1024 + bx * 128 + T.shift_right(tx, 6) * 64 + T.shift_right(T.bitwise_and(i, 15), 1) * 8 + T.bitwise_and(tx, 3) * 2 - 128:by * 131072 + T.shift_right(T.bitwise_and(tx, 63), 5) * 65536 + T.shift_right(i, 4) * 16384 + T.bitwise_and(i, 1) * 8192 + T.shift_right(T.bitwise_and(tx, 31), 2) * 1024 + bx * 128 + T.shift_right(tx, 6) * 64 + T.shift_right(T.bitwise_and(i, 15), 1) * 8 + T.bitwise_and(tx, 3) * 2 - 128 + 2] = C_local_1[i * 2:i * 2 + 2]
