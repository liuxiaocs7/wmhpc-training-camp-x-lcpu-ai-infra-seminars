                    T.mbarrier_expect_tx(mbarrier_1[k % 3], 16384)

                        for i in T.unroll(4):
                            T.ptx_ldmatrix(T.bool(True), 4, T.tvm_access_ptr(T.type_annotation("bfloat16"), B_shared.data, k % 3 * 8192 + tx // 64 % 2 * 4096 + ki * 1024 + (tx % 16 + ((i + tx // 16 % 2 * 8 // 16) // 2 + tx // 4 % 2) % 2 * 32 // 64) // 8 * 512 + (tx % 16 * 64 + ((i + tx // 16 % 2 * 8 // 16) // 2 + tx // 4 % 2) % 2 * 32 + (tx // 2 % 2 + i % 2) % 2 * 16 + (tx // 16 % 2 + tx % 2) % 2 * 8) % 512, 8, 1), T.tvm_access_ptr(T.type_annotation("bfloat16"), B_local.data, i * 8, 8, 2))
