@I.ir_module
class Module:
    @T.prim_func
    def gemm(A_handle: T.handle, B_handle: T.handle, C_handle: T.handle):
        T.func_attr({"target": T.target({"arch": "sm_100a", "host": {"keys": ["cpu"], "kind": "c", "tag": ""}, "keys": ["cuda", "gpu"], "kind": "cuda", "max_num_threads": 1024, "tag": "", "thread_warp_size": 32}), "tl.has_tma": T.bool(True), "tl.smem_alignment_map": {"A_shared": 1024, "B_shared": 1024}, "tl_tiled_ws_applied": 1})
        A = T.match_buffer(A_handle, (1024, 1024), "bfloat16", strides=(1024, 1))
        B = T.match_buffer(B_handle, (1024, 1024), "bfloat16", strides=(1024, 1))
        C = T.match_buffer(C_handle, (1024, 1024), strides=(1024, 1))
        with T.sblock("root"):
            T.reads()
            T.writes()
            A_shared = T.Buffer((3, 1, 16, 512), "bfloat16", scope="shared.dyn")
            B_shared = T.Buffer((3, 2, 8, 512), "bfloat16", scope="shared.dyn")
            C_local = T.Buffer((128,), scope="local")
            T.sblock_attr({"layout_map": {A_shared: metadata["tl.Layout"][0], B_shared: metadata["tl.Layout"][1], C_local: metadata["tl.Fragment"][0]}})
            bx = T.launch_thread("blockIdx.x", 8)
            by = T.launch_thread("blockIdx.y", 8)
            tx = T.launch_thread("threadIdx.x", 256)
            ty = T.launch_thread("threadIdx.y", 1)
            tz = T.launch_thread("threadIdx.z", 1)
            with T.sblock("tilelang_root"):
                T.reads()
                T.writes()
                mbarrier = T.handle("uint64", "shared.barrier")
                T.sblock_attr({"barrier_init": {mbarrier: [1, 1, 1, 128, 128, 128]}, "layout_map": {A_shared: metadata["tl.Layout"][0], B_shared: metadata["tl.Layout"][1], C_local: metadata["tl.Fragment"][0]}})
                A_shared = T.sblock_alloc_buffer((3, 1, 16, 512), "bfloat16", data=A_shared.data, scope="shared.dyn")
                B_shared = T.sblock_alloc_buffer((3, 2, 8, 512), "bfloat16", data=B_shared.data, scope="shared.dyn")
                C_local = T.sblock_alloc_buffer((128,), data=C_local.data, scope="local")
                mbarrier_1 = T.sblock_alloc_buffer((6,), "uint64", data=mbarrier, scope="shared.barrier")
                T.attr([128, 128], "kWarpSpecializationScope", 0)
                if tx < 128:
                    for k in range(16):
                        T.mbarrier_wait_parity(mbarrier_1[k % 3 + 3], T.bitwise_xor(k % 6 // 3, 1))
                        if T.tl_shuffle_elect(128):
                            T.mbarrier_expect_tx(mbarrier_1[k % 3], T.int64(16384))
                            T.tma_load(T.create_tma_descriptor(9, 2, A.data, 1024, 1024, T.int64(2), T.int64(2048), 64, 128, 1, 1, 0, 3, 2, 0), mbarrier_1[k % 3], T.tvm_access_ptr(T.type_annotation("bfloat16"), A_shared.data, k % 3 * 8192, 8192, 2), k * 64, by * 128, 0)
                        if T.tl_shuffle_elect(128):
                            T.mbarrier_expect_tx(mbarrier_1[k % 3], T.int64(16384))
                            for i in T.unroll(2):
                                T.tma_load(T.create_tma_descriptor(9, 2, B.data, 1024, 1024, T.int64(2), T.int64(2048), 64, 64, 1, 1, 0, 3, 2, 0), mbarrier_1[k % 3], T.tvm_access_ptr(T.type_annotation("bfloat16"), B_shared.data, k % 3 * 8192 + i * 4096, 4096, 2), bx * 128 + i * 64, k * 64, 0)
                            T.ptx_arrive_barrier(mbarrier_1[k % 3])
                else:
                    for i in T.unroll(32, annotations={"pragma_unroll_explicit": T.bool(False)}):
                        for vec in T.vectorized(4):
                            C_local[i * 4 + vec] = T.float32(0.0)
                    for k in range(16):
                        T.mbarrier_wait_parity(mbarrier_1[k % 3], k % 6 // 3)
                        with T.sblock("_gemm_ssr"):
                            T.reads()
                            T.writes()
                            T.sblock_attr({"lexical_alloc_scope": 1})
                            A_local = T.sblock_alloc_buffer((32,), "bfloat16", scope="local")
                            B_local = T.sblock_alloc_buffer((32,), "bfloat16", scope="local")
                            for ki in range(4):
                                for i in T.unroll(4, annotations={"pragma_unroll_explicit": False}):
                                    T.ptx_ldmatrix(T.bool(False), 4, T.tvm_access_ptr(T.type_annotation("bfloat16"), A_shared.data, k % 3 * 8192 + tx % 64 // 32 * 4096 + i * 1024 + (tx % 16 * 64 + ((ki * 16 + tx % 32 // 16 * 8) // 32 + tx % 8 // 4) % 2 * 32) // 512 * 512 + (tx % 16 * 64 + ((ki * 16 + tx % 32 // 16 * 8) // 32 + tx % 8 // 4) % 2 * 32 + (tx % 4 // 2 + ki % 2) % 2 * 16 + (tx % 32 // 16 + tx % 2) % 2 * 8) % 512, 8, 1), T.tvm_access_ptr(T.type_annotation("bfloat16"), A_local.data, i * 8, 8, 2))
                                for i in T.unroll(4, annotations={"pragma_unroll_explicit": False}):
                                    T.ptx_ldmatrix(T.bool(True), 4, T.tvm_access_ptr(T.type_annotation("bfloat16"), B_shared.data, k % 3 * 8192 + tx % 128 // 64 * 4096 + ki * 1024 + (tx % 16 * 64 + ((i * 16 + tx % 32 // 16 * 8) // 32 + tx % 8 // 4) % 2 * 32) // 512 * 512 + (tx % 16 * 64 + ((i * 16 + tx % 32 // 16 * 8) // 32 + tx % 8 // 4) % 2 * 32 + (tx % 4 // 2 + i % 2) % 2 * 16 + (tx % 32 // 16 + tx % 2) % 2 * 8) % 512, 8, 1), T.tvm_access_ptr(T.type_annotation("bfloat16"), B_local.data, i * 8, 8, 2))
                                for i, j in T.grid(4, 4):
                                    T.ptx_mma("float32", "m16n8k16", "row", "col", "bf16", "bf16", "fp32", A_local.data, i * 8, B_local.data, j * 8, C_local.data, i * 32 + j * 8, T.bool(False))
                                    T.ptx_mma("float32", "m16n8k16", "row", "col", "bf16", "bf16", "fp32", A_local.data, i * 8, B_local.data, j * 8 + 4, C_local.data, i * 32 + j * 8 + 4, T.bool(False))
                        T.ptx_arrive_barrier(mbarrier_1[k % 3 + 3])
                    for i in T.unroll(64, annotations={"pragma_unroll_explicit": T.bool(False)}):
                        for vec in T.vectorized(2):
                            C[by * 128 + tx % 64 // 32 * 64 + i // 16 * 16 + i % 2 * 8 + tx % 32 // 4, bx * 128 + tx // 64 * 64 + i % 16 // 2 * 8 + tx % 4 * 2 + vec - 128] = C_local[i * 2 + vec]
